3: export function validateContact(body, now = Date.now()) {
4:   if (!body || typeof body !== 'object' || Array.isArray(body)) return { error: 'Pedido inválido.' };
5:   if (body.website) return { error: 'Não foi possível validar o pedido.' };
6:   const name = typeof body.name === 'string' ? body.name.trim() : '';
7:   const email = typeof body.email === 'string' ? body.email.trim() : '';
8:   const message = typeof body.message === 'string' ? body.message.trim() : '';
9:   if (name.length < 2 || name.length > 120 || /[\r\n]/.test(name)) return { error: 'Indique um nome válido.' };
10:   if (email.length > 254 || !/^[^\s@<>]+@[^\s@<>]+\.[^\s@<>]+$/.test(email)) return { error: 'Indique um e-mail válido.' };
11:   if (message.length < 10 || message.length > 5000) return { error: 'A mensagem deve ter entre 10 e 5000 caracteres.' };
12:   if (!Number.isFinite(Number(body.startedAt)) || Number(body.startedAt) > now - 1200 || Number(body.startedAt) < now - 86400000) return { error: 'Atualize a página e tente novamente.' };
13:   return { value: { name, email, message } };
14: }

16: export async function sendContact(data, options = {}) {
17:   const environment = options.environment || process.env;
18:   const fetcher = options.fetcher || fetch;
19:   if (!environment.RESEND_API_KEY || !environment.CONTACT_FROM_EMAIL) return { status: 503, message: 'O envio pelo formulário ainda não está disponível.' };
20:   try {
21:     const response = await fetcher('https://api.resend.com/emails', {
22:       method: 'POST', headers: { Authorization: `Bearer ${environment.RESEND_API_KEY}`, 'Content-Type': 'application/json' },
23:       body: JSON.stringify({ from: environment.CONTACT_FROM_EMAIL, to: [environment.CONTACT_TO_EMAIL || site.email], reply_to: data.email, subject: `Contacto Dipanda — ${data.name}`, text: `Nome: ${data.name}\nE-mail: ${data.email}\n\n${data.message}` }), signal: AbortSignal.timeout(15000)
24:     });
25:     const result = await response.json().catch(() => ({}));
26:     if (!response.ok || typeof result.id !== 'string') return { status: 502, message: 'Não foi possível confirmar o envio da mensagem. Tente mais tarde ou contacte-nos por e-mail.' };
27:     return { status: 200, message: 'A sua mensagem foi enviada. A equipa Dipanda entrará em contacto consigo.' };
28:   } catch { return { status: 502, message: 'Não foi possível confirmar o envio da mensagem. Contacte-nos por e-mail antes de repetir o pedido.' }; }
29: }
